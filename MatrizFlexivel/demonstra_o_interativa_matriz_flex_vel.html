Matriz Flexível (Lista Ortogonal 2D)

Células alocadas individualmente na memória, conectadas por ponteiros duplos horizontais e verticais.

Status

Matriz 1x1 inicializada.
void adicionarLinha() {
Celula* atual = head;
// Ir para a ultima linha
while (atual->inf != NULL) atual = atual->inf;
// Criar cabeca da nova linha
Celula* novaLinha = criarCelula(rand());
atual->inf = novaLinha;
novaLinha->sup = atual;
Celula* ant_nova = novaLinha;
Celula* topo_atual = atual->dir;
// Percorrer colunas costurando ponteiros
while (topo_atual != NULL) {
Celula* nova = criarCelula(rand());
// Costura horizontal (Esq/Dir)
ant_nova->dir = nova;
nova->esq = ant_nova;
// Costura vertical (Sup/Inf)
topo_atual->inf = nova;
nova->sup = topo_atual;
ant_nova = nova;
topo_atual = topo_atual->dir;
}
}
void adicionarColuna() {
Celula* atual = head;
// Ir para a ultima coluna
while (atual->dir != NULL) atual = atual->dir;
// Criar cabeca da nova coluna
Celula* novaCol = criarCelula(rand());
atual->dir = novaCol;
novaCol->esq = atual;
Celula* ant_nova = novaCol;
Celula* esq_atual = atual->inf;
// Percorrer linhas costurando ponteiros
while (esq_atual != NULL) {
Celula* nova = criarCelula(rand());
// Costura vertical (Sup/Inf)
ant_nova->inf = nova;
nova->sup = ant_nova;
// Costura horizontal (Esq/Dir)
esq_atual->dir = nova;
nova->esq = esq_atual;
ant_nova = nova;
esq_atual = esq_atual->inf;
}
